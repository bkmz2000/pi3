from graphics import *
import math, random

size(500, 500)

def make_ship_sprite():
    s = create_sprite(16, 16)
    for (x, y) in [(8,1),(9,4),(7,4),(12,13),(4,13),(8,10)]:
        set_pixel(s, x, y, Colors.white)
    return s

def make_asteroid_sprite():
    s = create_sprite(24, 24)
    cx, cy, r = 12, 12, 10
    for a_deg in range(0, 360, 3):
        a = math.radians(a_deg)
        rr = r + random.uniform(-2, 2)
        set_pixel(s, round(cx + math.cos(a) * rr), round(cy + math.sin(a) * rr), Colors.silver)
    return s

SHIP = make_ship_sprite()
ASTEROID = make_asteroid_sprite()

ship_x, ship_y = 250, 250
ship_vx, ship_vy = 0.0, 0.0
ship_angle = 0
GRAVITY = 0.05
THRUST = 0.15

bullets = []
asteroids = [
    {"x": random.uniform(0, 500), "y": random.uniform(0, 150),
     "vx": random.uniform(-1, 1), "vy": random.uniform(0.2, 0.8)}
    for _ in range(5)
]

def draw():
    global ship_x, ship_y, ship_vx, ship_vy, ship_angle

    if Keyboard.arrow_left.pressed:
        ship_angle -= 4
    if Keyboard.arrow_right.pressed:
        ship_angle += 4
    if Keyboard.arrow_up.pressed:
        rad = math.radians(ship_angle - 90)
        ship_vx += math.cos(rad) * THRUST
        ship_vy += math.sin(rad) * THRUST

    ship_vy += GRAVITY
    ship_x = (ship_x + ship_vx) % width()
    ship_y = (ship_y + ship_vy) % height()

    if Keyboard.space.pressed:
        rad = math.radians(ship_angle - 90)
        bullets.append({"x": ship_x, "y": ship_y,
                         "vx": math.cos(rad) * 6, "vy": math.sin(rad) * 6, "life": 40})
    for b in bullets:
        b["x"] += b["vx"]; b["y"] += b["vy"]; b["life"] -= 1
    bullets[:] = [b for b in bullets if b["life"] > 0]

    for a in asteroids:
        a["x"] = (a["x"] + a["vx"]) % width()
        a["y"] = (a["y"] + a["vy"]) % height()

    background(5, 5, 15)
    for a in asteroids:
        image(ASTEROID, a["x"] - 12, a["y"] - 12)

    fill(255, 220, 100)
    no_stroke()
    for b in bullets:
        circle(b["x"], b["y"], 2)

    push()
    translate(ship_x, ship_y)
    rotate(math.radians(ship_angle))
    image(SHIP, -8, -8)
    pop()

run(draw)
