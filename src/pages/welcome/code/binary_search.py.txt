from pi3 import debug

def binary_search(arr, target):
    lo, hi = 0, len(arr) - 1
    while lo <= hi:
        mid = (lo + hi) // 2
        debug.array(arr,
            red=debug.between(lo, hi, name="search window"),
            green=debug.named(mid, "mid"),
            stroke_blue=debug.singles(lo, hi, name="lo, hi"))
        debug.show()
        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            lo = mid + 1
        else:
            hi = mid - 1
    return -1

binary_search([4, 17, 34, 50, 60, 61, 67, 95, 121, 122, 139, 140,
               141, 142, 149, 152, 155, 156, 161, 184], 142)
