N = len(path)


def dft(points, keep):
    """Discrete Fourier transform — decompose a traced signature into rotating
    circles (epicycles). Each circle's radius and speed come straight out of
    this decomposition, nothing hand-tuned. Returns the `keep` strongest ones.
    """
    n = len(points)
    result = []
    for k in range(n):
        c = sum(points[i] * cmath.exp(-2j * cmath.pi * k * i / n) for i in range(n)) / n
        result.append((k, c))
    result.sort(key=lambda kc: -abs(kc[1]))
    return result[:keep]


coeffs = dft(path, 15)

trail = []
t = 0

def draw():
    global t, trail
    background(10, 15, 20)

    pos = complex(200, 200)  # canvas center
    stroke(90, 130, 125)
    stroke_width(1)
    no_fill()
    for k, c in coeffs:
        prev = pos
        pos += c * cmath.exp(2j * cmath.pi * k * t / N)
        if abs(c) > 1:
            circle(prev.real, prev.imag, abs(c))
            line(prev.real, prev.imag, pos.real, pos.imag)

    trail.append(pos)
    if len(trail) > N:
        trail.pop(0)

    stroke(246, 165, 96)
    stroke_width(2)
    for i in range(1, len(trail)):
        a, b = trail[i - 1], trail[i]
        line(a.real, a.imag, b.real, b.imag)

    no_stroke()
    fill(255, 255, 255)
    circle(pos.real, pos.imag, 3)

    t = (t + 1) % N

run(draw)
